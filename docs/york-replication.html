1.3 LHIN

Show code 
tbls[["lhin-summary"]] <- lhin |> 
  mutate(
    across(
      matches("amount|cmha|css"), 
      ~ real_dollars(.x, data_year, 2023)
    )
  ) |> 
  left_join(
    fir_data |> 
      filter(
        ut_name == "York R",
      ) |> 
      distinct(data_year = marsyear, pop_york = population, ut_name),
    by = c("data_year")
  ) |> 
  filter(serves_york) |> 
  mutate(
    amount_pc_gap = amount_pc - amount_pc_on,
    total_gap = amount_pc_gap * pop_york
  ) %>%
  bind_rows(
    summarize(.,
      lhin = "Central",
      data_year = Inf,
      across(
        c(matches("amount_pc$")),
        ~ weighted.mean(.x, pop_lhin, na.rm = TRUE)
      ), 
      across(
        c(matches("amount_pc_on$")),
        ~ weighted.mean(.x, pop_on, na.rm = TRUE)
      ), 
      across(
        c(matches("pop_(lhin|york)")),
        ~ mean(.x, na.rm = TRUE)
      ), 
      .by = c(ut_name)
    )  |> 
    mutate(
      amount_pc_gap = amount_pc - amount_pc_on,
      total_gap = amount_pc_gap * pop_york
    )
  ) |> 
  mutate(
    channel = "Local Health Integration Networks (LHINs)",
    data_year = data_year, 
    data_range = glue::glue("({min(data_year)}-{max(data_year[data_year < Inf])})"),
    headline_gap_pc = amount_pc_gap,
    headline_gap_total = total_gap
  )

tbls[["efis-summary"]] <- efis |> 
  filter(
    school_board |> stringr::str_detect("york")
  ) |> 
  left_join(
    fir_data |> 
      filter(
        ut_name == "York R",
      ) |> 
      distinct(data_year = marsyear, population, ut_name),
    by = c("data_year")
  ) |>  
  mutate(
    rev_provincial_pc_gap = rev_provincial_total_gap / population
  ) %>%
  bind_rows(
    summarize(., 
      data_year = Inf,
      across(
        c(matches("rev_provincial_ps$")),
        ~ weighted.mean(.x, avg_daily_enrolment, na.rm = TRUE)
      ), 
      across(
        c(matches("rev_provincial_ps_avg$")),
        ~ weighted.mean(.x, avg_daily_enrolment_ontario, na.rm = TRUE)
      ), 
      across(
        c(matches("^(avg_daily_enrolment|population)")),
        ~ mean(.x, na.rm = TRUE)
      ),
      .by = c(school_board, board_sector, ut_name)
    ) |> 
    mutate(
      rev_provincial_ps_gap = rev_provincial_ps - rev_provincial_ps_avg,
      rev_provincial_total_gap = rev_provincial_ps_gap * avg_daily_enrolment,
      rev_provincial_pc_gap = rev_provincial_total_gap / population,
    )
  ) |> 
  mutate(
    channel = c(
      "Public" = "Public School Boards",
      "Catholic" = "Roman Catholic Separate School Boards"
    )[board_sector],
    data_year = data_year, 
    data_range = glue::glue("({min(data_year)}-{max(data_year[data_year < Inf])})"),
    headline_gap_pc = rev_provincial_pc_gap,
    headline_gap_total = rev_provincial_total_gap
  )

tbls[["combined-gap-detail"]] <- tbls[c(
  "fir-summary-core",
  "fir-summary-soc",
  "t3010-summary",
  "lhin-summary",
  "efis-summary"
)] |> 
  map_dfr(\(.tbl) {
    .tbl |>
      dplyr::filter(
        ut_name == "York R"
      ) |>  
      dplyr::select(
        ut_name,
        channel,
        data_year,
        data_range,
        headline_gap_pc,
        headline_gap_total
      ) |> 
      dplyr::mutate(
        observed_at = dplyr::case_when(
          data_year == Inf ~ "Annual Average",
          data_year == min(data_year[data_year < Inf]) ~ "Start Year",
          data_year == max(data_year[data_year < Inf]) ~ "End Year",
          TRUE ~ NA_character_
        ) |> 
          factor(
            levels = c("Start Year", "End Year", "Annual Average")
          )
      ) |> 
      dplyr::filter(!is.na(observed_at)) |> 
      dplyr::select(-data_year)
  }) %>%
  dplyr::bind_rows(
    summarize(., 
      channel = "Z) Total",
      data_range = "(2015-2024)",
      headline_gap_pc = sum(headline_gap_pc, na.rm = TRUE),
      headline_gap_total = sum(headline_gap_total, na.rm = TRUE),
      .by = c(observed_at, ut_name)
    )
  ) |> 
  tidyr::pivot_wider(
    names_from = observed_at,
    values_from = c(headline_gap_pc, headline_gap_total),
    names_glue = "{observed_at}::{.value}"
  ) |> 
  dplyr::select(
    ut_name,
    channel, 
    data_range, 
    matches("Start"),
    matches("End"), 
    matches("Annual")
  )
