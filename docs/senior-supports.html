library(tidyverse)
devtools::load_all()


d <- fir_data |> 
  filter(marsyear < 2023) |> 
  group_by(ut_number) |> 
  filter(min(population) > 5e4, service == "Senior Support", cost_payer == "ontario") |> 
  group_by(marsyear, ut_number, ut_name) |> 
  summarize(
    total_real = sum(amount), 
    population = first(population),
    per_capita_real = sum(amount_pc),
    per_capita_cd23 = sum(amount_pc_cd23)
  )

d |>

    marsyear, ut_name,
